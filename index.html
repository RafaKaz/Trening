        <!-- Wyświetlacz rozkazu ruchowego -->
        <div class="command-box" id="command-display">USTAW SIĘ I UDERZ</div>

    </div>

    <div class="controls">
        <button id="replay-last-btn" class="replay-btn" disabled>▶ Powtórka (0.5x)</button>
        <button id="live-cam-btn" class="secondary" style="display: none;">? Aparat na żywo</button>
        <button id="voice-btn">? Głos: WŁĄCZONY</button>
        <button id="handedness-btn" class="secondary">Gracz: Praworęczny</button>
        <button id="clear-memory-btn" class="secondary">? Wyczyść pamięć sesji</button>

    <script>
        // --- PARAMETRY KONFIGURACYJNE ---
        let IS_RIGHT_HANDED = true;          // true = praworęczny, false = leworęczny
        let VOICE_ENABLED = true;            // true = odczyt głosowy rozkazów
        const OUT_TO_IN_THRESHOLD = 0.03;    // Próg wykrywania ruchu na zewnątrz
        const CLIP_DURATION_MS = 30000;      // Czas nagrywania tymczasowego klipu w pamięci (30 s)

        // Rozkazy ruchowe (krótkie polecenia do wykonania ciałem)
        const MOVEMENT_COMMANDS = {
            strongOutToIn: "Łokieć do biodra! Ręce w dół!",
            mildOutToIn: "Opuść łokieć! Trzymaj plecy!",
            goodInToOut: "Trzymaj ten tor! Powtórz!"
        };

        const sourceVideo = document.getElementById('source-video');
        const canvasElement = document.getElementById('output-canvas');
        const canvasCtx = canvasElement.getContext('2d');
        const replayPlayer = document.getElementById('replay-player');

        const phaseIndicator = document.getElementById('phase-indicator');
        const feedbackDisplay = document.getElementById('feedback-display');
        const commandDisplay = document.getElementById('command-display');
        const sessionClipsInfo = document.getElementById('session-clips-info');

        const replayLastBtn = document.getElementById('replay-last-btn');
        const liveCamBtn = document.getElementById('live-cam-btn');
        const voiceBtn = document.getElementById('voice-btn');
        const handednessBtn = document.getElementById('handedness-btn');
        const clearMemoryBtn = document.getElementById('clear-memory-btn');

        let swingState = 'IDLE';
        let topWristX = 0;
        let topWristY = 1;
        let cooldown = false;

        let sharedStream = null;
        let mediaRecorder = null;
        let currentChunks = [];
        let isRecordingClip = false;
        let sessionClipUrls = [];
        let isAnalyzingFrame = false;

        // Odczytuje rozkaz ruchowy głosem
        function speak(commandText) {
            if (!VOICE_ENABLED || !('speechSynthesis' in window)) return;
            window.speechSynthesis.cancel();
            const utterance = new SpeechSynthesisUtterance(commandText);
            utterance.lang = 'pl-PL';
            utterance.rate = 1.15;
            window.speechSynthesis.speak(utterance);
        }

        function updateClipsInfo() {
            const count = sessionClipUrls.length;
            sessionClipsInfo.textContent = `Tymczasowe klipy (30 s) w tej sesji: ${count} (tylko w pamięci)`;
            replayLastBtn.disabled = count === 0;
        }

        // Nagrywa 30-sekundowy tymczasowy klip w pamięci RAM
        function captureTemporaryClip() {
            if (!sharedStream || isRecordingClip || typeof MediaRecorder === 'undefined') return;

            currentChunks = [];
            isRecordingClip = true;

            try {
                mediaRecorder = new MediaRecorder(sharedStream);
            } catch (e) {
                isRecordingClip = false;
                return;
            }

            mediaRecorder.ondataavailable = (event) => {
                if (event.data && event.data.size > 0) {
                    currentChunks.push(event.data);
                }
            };

            mediaRecorder.onstop = () => {
                isRecordingClip = false;
                if (currentChunks.length === 0) return;

                const blob = new Blob(currentChunks, { type: mediaRecorder.mimeType || 'video/webm' });
                const tempUrl = URL.createObjectURL(blob);
                sessionClipUrls.push(tempUrl);
                updateClipsInfo();
            };

            mediaRecorder.start();

            setTimeout(() => {
                if (mediaRecorder && mediaRecorder.state === 'recording') {
                    mediaRecorder.stop();
                }
            }, CLIP_DURATION_MS);
        }

        function showLastTemporaryReplay() {
            if (sessionClipUrls.length === 0) return;

            const latestUrl = sessionClipUrls[sessionClipUrls.length - 1];
            canvasElement.style.display = 'none';
            replayPlayer.style.display = 'block';
            replayPlayer.src = latestUrl;
            replayPlayer.playbackRate = 0.5;
            replayPlayer.play();

            replayLastBtn.style.display = 'none';
            liveCamBtn.style.display = 'inline-block';
            phaseIndicator.textContent = "Tryb powtórki 30 s (0.5x) — tylko w pamięci";
        }

        function returnToLiveCamera() {
            replayPlayer.pause();
            replayPlayer.removeAttribute('src');
            replayPlayer.load();
            replayPlayer.style.display = 'none';
            canvasElement.style.display = 'block';

            liveCamBtn.style.display = 'none';
            replayLastBtn.style.display = 'inline-block';
            phaseIndicator.textContent = "Aparat na żywo — gotowy na swing";
        }

        function clearTemporarySessionMemory() {
            returnToLiveCamera();
            sessionClipUrls.forEach((url) => URL.revokeObjectURL(url));
            sessionClipUrls = [];
            updateClipsInfo();
            feedbackDisplay.className = "feedback-text";
            feedbackDisplay.textContent = "Wyczyszczono pamięć sesji";
            commandDisplay.textContent = "USTAW SIĘ I UDERZ";
        }

        const pose = new Pose({
            locateFile: (file) => `https://cdn.jsdelivr.net/npm/@mediapipe/pose/${file}`
        });

        pose.setOptions({
            modelComplexity: 1,
            smoothLandmarks: true,
            minDetectionConfidence: 0.6,
            minTrackingConfidence: 0.6
        });

        pose.onResults((results) => {
            if (replayPlayer.style.display === 'block') return;

            canvasElement.width = results.image.width;
            canvasElement.height = results.image.height;

            canvasCtx.save();
            canvasCtx.clearRect(0, 0, canvasElement.width, canvasElement.height);
            canvasCtx.drawImage(results.image, 0, 0, canvasElement.width, canvasElement.height);

            if (results.poseLandmarks) {
                drawConnectors(canvasCtx, results.poseLandmarks, POSE_CONNECTIONS, { color: '#10b981', lineWidth: 3 });
                drawLandmarks(canvasCtx, results.poseLandmarks, { color: '#ef4444', lineWidth: 2, radius: 4 });
                analyzeSwing(results.poseLandmarks);
            } else {
                phaseIndicator.textContent = "Szukam sylwetki w kadrze...";
            }

            canvasCtx.restore();
        });

        function analyzeSwing(landmarks) {
            if (cooldown) return;

            const leftShoulder = landmarks[11];
            const rightShoulder = landmarks[12];
            const leftWrist = landmarks[15];
            const rightWrist = landmarks[16];

            const handsX = (leftWrist.x + rightWrist.x) / 2;
            const handsY = (leftWrist.y + rightWrist.y) / 2;
            const shouldersY = (leftShoulder.y + rightShoulder.y) / 2;

            if (handsY < shouldersY - 0.05) {
                if (swingState !== 'BACKSWING') {
                    swingState = 'BACKSWING';
                    topWristY = handsY;
                    topWristX = handsX;
                    phaseIndicator.textContent = "Faza: Zamach";
                }
                if (handsY < topWristY) {
                    topWristY = handsY;
                    topWristX = handsX;
                }
            }

            if (swingState === 'BACKSWING' && handsY > topWristY + 0.08) {
                swingState = 'DOWNSWING';
                phaseIndicator.textContent = "Nagrywanie 30 s w pamięci";
                captureTemporaryClip();

                const horizontalShift = IS_RIGHT_HANDED ? (handsX - topWristX) : (topWristX - handsX);
                evaluatePath(horizontalShift);
            }
        }

        // Wydaje odpowiedni rozkaz ruchowy na ekranie i głosowo
        function evaluatePath(shift) {
            cooldown = true;

            if (shift > OUT_TO_IN_THRESHOLD * 2) {
                feedbackDisplay.className = "feedback-text bad";
                feedbackDisplay.textContent = "Out-to-In (duże)";
                commandDisplay.style.borderLeftColor = "#ef4444";
                commandDisplay.textContent = MOVEMENT_COMMANDS.strongOutToIn;
                speak(MOVEMENT_COMMANDS.strongOutToIn);
            } else if (shift > OUT_TO_IN_THRESHOLD) {
                feedbackDisplay.className = "feedback-text bad";
                feedbackDisplay.textContent = "Out-to-In (lekkie)";
                commandDisplay.style.borderLeftColor = "#f59e0b";
                commandDisplay.textContent = MOVEMENT_COMMANDS.mildOutToIn;
                speak(MOVEMENT_COMMANDS.mildOutToIn);
            } else {
                feedbackDisplay.className = "feedback-text good";
                feedbackDisplay.textContent = "In-to-Out (OK)";
                commandDisplay.style.borderLeftColor = "#10b981";
                commandDisplay.textContent = MOVEMENT_COMMANDS.goodInToOut;
                speak(MOVEMENT_COMMANDS.goodInToOut);
            }

            setTimeout(() => {
                swingState = 'IDLE';
                cooldown = false;
                if (replayPlayer.style.display !== 'block') {
                    phaseIndicator.textContent = "Gotowy na swing";
                }
            }, 3500);
        }

        async function processVideoLoop() {
            if (
                sourceVideo.readyState >= 2 &&
                replayPlayer.style.display !== 'block' &&
                !isAnalyzingFrame
            ) {
                isAnalyzingFrame = true;
                try {
                    await pose.send({ image: sourceVideo });
                } finally {
                    isAnalyzingFrame = false;
                }
            }
            requestAnimationFrame(processVideoLoop);
        }

        async function initCameraAndAI() {
            if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {
                phaseIndicator.textContent = "Brak dostępu do aparatu";
                commandDisplay.textContent = "UŻYJ HTTPS LUB LOCALHOST";
                return;
            }

            try {
                sharedStream = await navigator.mediaDevices.getUserMedia({
                    video: {
                        facingMode: { ideal: 'environment' },
                        width: { ideal: 720 },
                        height: { ideal: 960 }
                    },
                    audio: false
                });

                sourceVideo.srcObject = sharedStream;
                await sourceVideo.play();

                phaseIndicator.textContent = "Aparat aktywny";
                commandDisplay.textContent = "WYKONAJ SWING";
                requestAnimationFrame(processVideoLoop);
            } catch (err) {
                phaseIndicator.textContent = "Błąd aparatu";
                commandDisplay.textContent = "SPRAWDŹ ZGODĘ I HTTPS";
            }
        }

        replayLastBtn.addEventListener('click', showLastTemporaryReplay);
        liveCamBtn.addEventListener('click', returnToLiveCamera);
        clearMemoryBtn.addEventListener('click', clearTemporarySessionMemory);

        voiceBtn.addEventListener('click', () => {
            VOICE_ENABLED = !VOICE_ENABLED;
            voiceBtn.textContent = VOICE_ENABLED ? "? Głos: WŁĄCZONY" : "? Głos: WYŁĄCZONY";
        });

        handednessBtn.addEventListener('click', () => {
            IS_RIGHT_HANDED = !IS_RIGHT_HANDED;
            handednessBtn.textContent = IS_RIGHT_HANDED ? "Gracz: Praworęczny" : "Gracz: Leworęczny";
        });

        window.addEventListener('DOMContentLoaded', initCameraAndAI);
        window.addEventListener('beforeunload', () => {
            sessionClipUrls.forEach((url) => URL.revokeObjectURL(url));
        });
    </script>
